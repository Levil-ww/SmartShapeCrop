### 1.2 修复：OCR边距自洽检测
```python
def _ocr_margins_self_consistent(total_w, total_h,
                                 ml_ocr, mr_ocr, mt_ocr, mb_ocr):
    """OCR边距值自洽吗？→ 内推的内框在合理范围就信任OCR"""
    # 水平方向：OCR左右和 → 内推inner_w
    inferred_inner_w = total_w - ml_ocr - mr_ocr
    # 垂直方向：OCR上下和 → 内推inner_h
    inferred_inner_h = total_h - mt_ocr - mb_ocr

    MIN_RATIO, MAX_RATIO = 0.10, 0.90  # 内框占外框比例合理范围

    h_valid = (MIN_RATIO * total_w < inferred_inner_w < MAX_RATIO * total_w)
    v_valid = (MIN_RATIO * total_h < inferred_inner_h < MAX_RATIO * total_h)

    if h_valid:
        logger.info(f"[OCR自洽] 水平边距OCR自洽（内推inner_w={inferred_inner_w:.1f}，"
                    f"占外框{inferred_inner_w/total_w:.0%}∈[10%,90%]），采用OCR值")
        margin_left, margin_right = ml_ocr, mr_ocr
    if v_valid:
        logger.info(f"[OCR自洽] 垂直边距OCR自洽，采用OCR值")
        margin_top, margin_bottom = mt_ocr, mb_ocr

    return h_valid, v_valid
```

**应用效果**：
- 78×58cm案例：OCR左12+右7.5=19.5 → 内推inner_w=58.5 → 58.5/78=75% ∈ [10%,90%] → **水平边距采用OCR值12和7.5 ✅**
- OCR不一致场景：自动退回几何路径，不破坏兜底。

## 二、问题B：左=2/右=47.5（应为左=12/右=7.5）

### 2.1 根因：OCR多位数拆分 + 异常值无拦截
```
OCR原始Token列表（按位置排序）：
  [(text="左", x=10), (text="1", x=30), (text="2", x=42),    ← "左 1 2" 被拆成3个Token!
   (text="右", x=110), (text="7", x=130), (text=".5", x=142)] ← "右 7 .5" 被拆成小数!
         │
         ▼ 旧逻辑：每个Token单独转数字
  左距候选：[2]（忽略了前面的1，因为"左"标签就近匹配最后一个数字）
  右距候选：[7] 或 [47.5（OCR把7读成47）]（噪声干扰）
```

### 2.2 修复①：相邻数字Token合并
```python
def _merge_adjacent_number_tokens(tokens_with_pos):
    """
    相邻数字Token合并：
    ["左","1","2"]       →  12
    ["7",".5"]           →  7.5
    ["3","6",".5"]       →  36.5
    合并规则：方向标签后紧跟的连续数字Token合并为单一数字
    """
    merged = []
    i = 0
    num_pattern = re.compile(r'(\d+\.?\d*|\.\d+)')  # 支持 .5 这种开头小数

    while i < len(tokens_with_pos):
        tok = tokens_with_pos[i]

        if num_pattern.match(tok.text):
            # 找到一个数字，尝试向后合并连续相邻数字
            combined_val = tok.text
            last_x = tok.x + tok.w  # 结束X坐标

            j = i + 1
            while j < len(tokens_with_pos):
                next_tok = tokens_with_pos[j]
                gap = next_tok.x - last_x
                if gap < tok.w * 0.8 and num_pattern.match(next_tok.text):
                    # 相邻间距<0.8个字符宽 → 是被拆分的同一数字
                    combined_val += next_tok.text
                    last_x = next_tok.x + next_tok.w
                    j += 1
                else:
                    break

            # 合并后转float
            try:
                merged.append(Token(float(combined_val), tok.x, last_x - tok.x))
            except ValueError:
                pass
            i = j
        else:
            # 非数字（方向标签等）→ 原样保留
            merged.append(tok)
            i += 1

    return merged
```

### 2.3 修复②：50%规则 + OCR几何交叉验证
```python
def _validate_pair_value(val, outer_max_side, geometry_val_cm,
                         outer_side_for_field):
    """边距值验证：50%上限 + 几何交叉校验"""
    # 规则1：边距 ≤ 50% × 外框最大边
    if val > outer_max_side * 0.50:
        logger.warning(f"[50%拦截] {val:.1f}cm > 外框最大边{outer_max_side:.1f}×50%，可疑，拒绝")
        return False, "50%规则"

    # 规则2：与几何值差>15% 且 >5cm → 触发聚焦OCR重扫
    if geometry_val_cm is not None:
        diff_pct = abs(val - geometry_val_cm) / max(geometry_val_cm, 0.01)
        diff_abs = abs(val - geometry_val_cm)
        if diff_pct > 0.15 and diff_abs > 5.0:
            logger.info(f"[交叉校验] {val:.1f} vs 几何{geometry_val_cm:.1f} "
                        f"差{diff_pct:.0%}/{diff_abs:.1f}cm，触发聚焦OCR重扫")
            rescanned = _focused_ocr_rescan(target_field)  # 针对该字段单独OCR
            if rescanned:
                return True, "rescanned", rescanned

    return True, "ok"

# 外框最大边 = 78 → 50%上限 = 39cm
# 47.5 > 39 → 50%规则拦截！❌ 该值被丢弃 → 采用合并后的7.5 ✅
```
